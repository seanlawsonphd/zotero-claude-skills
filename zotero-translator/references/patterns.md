# Translator code patterns (tested against RT.com and TASS.com, Zotero 10)

Copy these rather than re-deriving them. Everything runs inside Zotero's translator sandbox, where `Zotero`, `Z`, `ZU`, `attr()`, `text()`, `innerText()`, `requestDocument()` and friends are globals (see `index.d.ts` in the repo). Use tabs for indentation; the repo's ESLint config enforces it.

function detectWeb(doc, url) {

		}

## Pick-list: generic anchor scan

Works on homepages, section fronts, search results, tag and topic pages without per-page selectors. The three traps it handles: card links that wrap headline + summary + date (prefer the headline element), lazy-image links whose `<noscript>` fallback appears as literal `<img …>` text (strip `noscript`), and story links inside hidden mega-menus or sticky feeds that exist on every page (exclude their containers, found via the ancestry dump in `site-analysis.md`).

```js
function getSearchResults(doc, checkOnly) {
	var items = {};
	var found = false;
	var rows = doc.querySelectorAll('a[href]');
	for (let row of rows) {
		let href = row.href;
		if (!ARTICLE_URL_RE.test(href) || items[href]) continue;
		// Skip the hidden "Sections" mega-menu and the news-feed panel, which
		// preview stories on every page
		if (row.closest('.menu__content, .news-feed, header')) continue;
		// Story cards wrap the headline, lead and date in a single link, so
		// prefer the headline element when there is one
		let title = linkText(row.querySelector('.news-content__title, .news-preview__title') || row);
		if (!title) continue;
		if (checkOnly) return true;
		found = true;
		items[href] = title;
	}
	return found ? items : false;
}

// Text of a link without its <noscript> fallbacks, which browsers expose as
// raw markup in textContent (lazy-loaded images are wrapped that way)
function linkText(link) {
	let clone = link.cloneNode(true);
	for (let noscript of clone.querySelectorAll('noscript')) {
		noscript.remove();
	}
	return ZU.trimInternal(clone.textContent);
}
```

`row.href` is absolute in a browser, so match the full URL. The `items[href]` check keeps the first title per URL; image-only links come first on many sites, which is why empty-after-noscript text must be skipped before the dedupe records anything.

## Scrape: delegate to Embedded Metadata, then fix fields

```js
async function scrape(doc, url = doc.location.href) {
	let itemType = detectWeb(doc, url) || 'newspaperArticle';
	let linkedData = getLinkedData(doc);

	let translator = Zotero.loadTranslator('web');
	translator.setTranslator(EM_TRANSLATOR_ID);
	translator.setDocument(doc);

	translator.setHandler('itemDone', (_obj, item) => {
		let headline = (linkedData && linkedData.headline) || text(doc, 'h1.article__heading');
		if (headline) item.title = ZU.trimInternal(headline);

		let published = (linkedData && linkedData.datePublished)
			|| attr(doc, 'meta[name="published_time_telegram"]', 'content');
		if (published) item.date = ZU.strToISO(published);

		// The page metadata only credits the organization, so take the
		// authors from the byline instead (empty for unsigned agency copy)
		item.creators = getAuthors(doc);

		// The standfirst under the headline makes a better abstract than the
		// meta description
		let summary = text(doc, '.article__summary');
		if (summary) item.abstractNote = ZU.trimInternal(summary);

		let canonical = linkedData && linkedData.mainEntityOfPage && linkedData.mainEntityOfPage['@id'];
		item.url = canonical
			|| attr(doc, 'meta[property="og:url"]', 'content')
			|| url.replace(/[?#].*/, '');

		if (!item.language) item.language = 'en';

		if (itemType == 'tvBroadcast') {
			// Breadcrumbs read Home > Shows > <Program>
			let crumbs = doc.querySelectorAll('.breadcrumbs__links');
			let program = crumbs.length ? ZU.trimInternal(crumbs[crumbs.length - 1].textContent) : '';
			if (program && program != 'Shows') item.programTitle = program;
			item.network = 'RT';
			delete item.publicationTitle; // not a valid field for tvBroadcast
		}
		else {
			item.publicationTitle = 'RT';
			let section = attr(doc, 'meta[name="article:section"]', 'content');
			if (section) item.section = ZU.trimInternal(section);
		}

		// Merge EM's tags (keywords meta) with the page's topic tags
		let tags = new Set();
		for (let tag of item.tags) {
			tags.add(typeof tag == 'string' ? tag : tag.tag);
		}
		for (let link of doc.querySelectorAll('.article a[rel="tag"]')) {
			let tag = ZU.trimInternal(link.textContent);
			if (tag) tags.add(tag);
		}
		item.tags = Array.from(tags, tag => ({ tag }));

		item.complete();
	});

	let em = await translator.getTranslatorObject();
	em.itemType = itemType;
	await em.doWeb(doc, url);
}
```

Setting `em.itemType` before `doWeb` is what makes EM produce a `newspaperArticle` instead of guessing `webpage`. If no override is needed, `await translator.translate()` is enough.

## JSON-LD helpers (handle arrays and `@graph`)

```js
// Every schema.org entry on the page, with @graph containers flattened
function getLinkedDataEntries(doc) {
	let entries = [];
	for (let script of doc.querySelectorAll('script[type="application/ld+json"]')) {
		try {
			let data = JSON.parse(script.textContent);
			for (let entry of [].concat(data)) {
				if (!entry) continue;
				entries.push(...[].concat(entry['@graph'] || entry));
			}
		}
		catch (e) {
			// skip malformed blocks
		}
	}
	return entries;
}

function getLinkedData(doc) {
	return getLinkedDataEntries(doc).find(entry => entry && entry['@type'] == 'NewsArticle') || null;
}
```

## Section from navigation JSON-LD with a `<title>` fallback (TASS)

TASS labels sections nowhere on the article itself, but its navigation is in a `SiteNavigationElement` graph and the `<title>` ends in " - Section - TASS".

```js
function getSection(doc, url) {
	let segment = url.match(/^https?:\/\/(?:www\.)?tass\.com\/([a-z-]+)\/\d/i);
	if (segment) {
		for (let entry of getLinkedDataEntries(doc)) {
			if (entry && entry['@type'] == 'SiteNavigationElement' && entry.url && entry.name
				&& entry.url.replace(/\/+$/, '').endsWith('/' + segment[1])) {
				return ZU.trimInternal(entry.name);
			}
		}
	}
	let parts = doc.title.split(/\s+-\s+/);
	if (parts.length >= 3) return ZU.trimInternal(parts[parts.length - 2]);
	return '';
}
```

## Bylines: bold name after "By", with a role suffix

RT prints `<p><em>By </em><strong><em>Elizabeth Blade</em></strong><em>, RT Middle East correspondent</em></p>` at the end of signed pieces; news stories have no byline at all.

```js
function getAuthors(doc) {
	let byline = doc.querySelector('.article__author-text');
	if (!byline) return [];
	let names = Array.from(byline.querySelectorAll('strong'), el => ZU.trimInternal(el.textContent))
		.filter(Boolean);
	if (!names.length) {
		let match = ZU.trimInternal(byline.textContent).match(/^By\s+([^,]+)/i);
		if (match) names = [match[1]];
	}
	let creators = [];
	for (let joined of names) {
		for (let author of joined.replace(/^By\s+/i, '').split(/\s*(?:,|\band\b|&)\s*/)) {
			author = author.trim();
			if (!author || /^(RT|Russia Today)\b/i.test(author)) continue;
			creators.push(ZU.cleanAuthor(author, 'author'));
		}
	}
	return creators;
}
```

`ZU.trimInternal` also collapses non-breaking spaces (TASS-style `&nbsp;Name`). Drop organization names so they never become a person.

## Field conventions worth keeping

- `date`: date-only ISO via `ZU.strToISO`; the JSON-LD timestamp is UTC.
- `libraryCatalog`: leave EM's hostname default.
- `shortTitle`: generated by the framework from "Title: subtitle"; official translators keep it.
- `language`: EM reads `<html lang>` or `meta[name=language]`; set a default when the site declares none (TASS).
- `attachments`: EM adds `{ title: "Snapshot", document: doc }`; keep it. The Connector only saves it for single-item saves.
- Validity: `section`/`publicationTitle` are invalid for `tvBroadcast`; `programTitle`/`network` are invalid for `newspaperArticle`. Set per branch.
